# Dependências da automação SIGEF -> SEI.
#
# Instalar (uma vez só, num prompt de comando dentro desta pasta):
#
#     python -m pip install -r requirements.txt
#
# Não é preciso rodar "playwright install": a automação se conecta ao
# Google Chrome que já está instalado na máquina, em modo de depuração,
# em vez de usar um navegador próprio do Playwright.

pywebview>=5.0          # a janela do programa (app_janela.py)
openpyxl>=3.1           # lê as planilhas (modo lote do SEI e importação do DARE)
psutil>=5.9             # DARE: aba Desempenho (CPU, memória, disco e rede)
pywin32>=306; sys_platform == "win32"   # Excel aberto (BB e importação do DARE)
